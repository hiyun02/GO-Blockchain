# 0. 프로젝트 폴더로 이동

# ============================================================
# 1. Gov 부트노드 실행: 호스트 포트 5000
# ============================================================
docker run -d --name hs8-gov-0 --network hotstuff8 --label hs.stack=hotstuff8 --restart unless-stopped -p 5000:5000 -v hs8-gov-0-data:/data -e Gov_ID=Gov-A -e PORT=5000 -e NODE_ADDR=hs8-gov-0:5000 -e BOOTSTRAP_ADDR=hs8-gov-0:5000 -e Gov_DB_PATH=/data/blockchain_db -e BLOCK_HISTORY_PATH=/data/block_history.txt -e HOTSTUFF_BATCH_SIZE=1 -e HOTSTUFF_BATCH_TIMEOUT_SEC=2 -e HOTSTUFF_VIEW_TIMEOUT_SEC=8 --health-cmd "wget -qO- http://127.0.0.1:5000/status || exit 1" --health-interval 2s --health-timeout 2s --health-retries 30 hiyun2002/gov-node:hotstuff
docker inspect -f '{{.State.Health.Status}}' hs8-gov-0
docker logs --tail 100 hs8-gov-0
